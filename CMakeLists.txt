project(lmplot VERSION 0.2.1 LANGUAGES C CXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_CXX_EXTENSIONS OFF)
set(CMAKE_C_STANDARD 11)
set(CMAKE_C_EXTENSIONS ON)
if(NOT CMAKE_BUILD_TYPE)
  set(CMAKE_BUILD_TYPE Release)
endif()
# Rule 14: third-party code stays static; Linux uses the system glibc.
set(BUILD_SHARED_LIBS OFF)
if(CMAKE_SYSTEM_NAME STREQUAL "Linux")
  set(CMAKE_FIND_LIBRARY_SUFFIXES ".a")
endif()

set(GP third_party/gnuplot)
if(NOT EXISTS "${CMAKE_CURRENT_SOURCE_DIR}/${GP}/src/plot.c")
  message(FATAL_ERROR "third_party/gnuplot missing: run scripts/sync-gnuplot.sh")
endif()

# Embed the DSL schema (same scheme as lmhtml/lmcad).
set(LMPLOT_SCHEMA "${CMAKE_CURRENT_SOURCE_DIR}/schema/lmplot.dsl.json")
set_property(DIRECTORY APPEND PROPERTY CMAKE_CONFIGURE_DEPENDS "${LMPLOT_SCHEMA}")
file(READ "${LMPLOT_SCHEMA}" LMPLOT_SCHEMA_HEX HEX)
string(LENGTH "${LMPLOT_SCHEMA_HEX}" LMPLOT_SCHEMA_HEX_LENGTH)
math(EXPR LMPLOT_SCHEMA_LAST_BYTE "${LMPLOT_SCHEMA_HEX_LENGTH} - 2")
set(LMPLOT_SCHEMA_INITIALIZER "")
foreach(offset RANGE 0 ${LMPLOT_SCHEMA_LAST_BYTE} 2)
  string(SUBSTRING "${LMPLOT_SCHEMA_HEX}" ${offset} 2 byte)
  string(APPEND LMPLOT_SCHEMA_INITIALIZER "0x${byte},")
endforeach()
set(GEN "${CMAKE_CURRENT_BINARY_DIR}/generated")
file(MAKE_DIRECTORY "${GEN}" "${GEN}/PostScript")
set(NOTICES "${CMAKE_CURRENT_SOURCE_DIR}/THIRD_PARTY_NOTICES")
set_property(DIRECTORY APPEND PROPERTY CMAKE_CONFIGURE_DEPENDS "${NOTICES}")
file(READ "${NOTICES}" NOTICES_HEX HEX)
string(REGEX REPLACE "([0-9a-f][0-9a-f])" "0x\\1," NOTICES_HEX "${NOTICES_HEX}")
file(WRITE "${CMAKE_CURRENT_BINARY_DIR}/generated/notices_embedded.hpp"
  "#pragma once\n#include <cstddef>\nnamespace lmplot { inline constexpr unsigned char embedded_notices[] = {${NOTICES_HEX}0}; inline constexpr std::size_t embedded_notices_size = sizeof(embedded_notices) - 1; }\n")
file(WRITE "${GEN}/schema_embedded.hpp"
  "#pragma once\n#include <cstddef>\nnamespace lmplot { inline constexpr unsigned char embedded_schema[] = {${LMPLOT_SCHEMA_INITIALIZER}}; inline constexpr std::size_t embedded_schema_size = sizeof(embedded_schema); }\n")

# post.trm embeds its PostScript prologues from PostScript/prologues.h when
# GNUPLOT_PS_DIR is undefined; upstream generates it with a shell script, so
# do the same transformation here (quote, tab, line-end escapes).
file(GLOB PS_FILES "${CMAKE_CURRENT_SOURCE_DIR}/${GP}/term/PostScript/*.ps")
list(SORT PS_FILES COMPARE NATURAL)
set(PROLOGUES "/* generated by CMakeLists.txt */\n")
foreach(ps ${PS_FILES})
  get_filename_component(base "${ps}" NAME_WE)
  string(REPLACE "-" "_" ident "${base}")
  file(READ "${ps}" body)
  string(REPLACE "\"" "\\\"" body "${body}")
  string(REPLACE "\t" "\\t" body "${body}")
  string(REGEX REPLACE "\n$" "" body "${body}")
  string(REPLACE "\n" "\\n\",\n\"" body "${body}")
  string(APPEND PROLOGUES "static const char *prologue_${ident}_ps[] = {\n\"${body}\\n\",\nNULL\n};\n\n")
endforeach()
file(WRITE "${GEN}/PostScript/prologues.h" "${PROLOGUES}")

# version.c wants a build date; use the pinned commit's date so builds are reproducible.
file(WRITE "${GEN}/timestamp.h"
  "#ifndef GNUPLOT_TIMEBASE_H_INCLUDED\n#define GNUPLOT_TIMEBASE_H_INCLUDED\nconst char gnuplot_date[] = \"2023-09-08\";\n#endif\n")

# Embedded DejaVu fonts (see vendor/fonts/LICENSE-DejaVu) for the native png terminal.
function(embed_font var file symbol)
  file(READ "${CMAKE_CURRENT_SOURCE_DIR}/vendor/fonts/${file}" hex HEX)
  string(REGEX REPLACE "([0-9a-f][0-9a-f])" "0x\\1," hex "${hex}")
  set(${var} "${${var}}const unsigned char ${symbol}[] = {${hex}0};\n" PARENT_SCOPE)
endfunction()
set(FONT_C "")
embed_font(FONT_C DejaVuSans.ttf lmr_font_sans)
embed_font(FONT_C DejaVuSans-Bold.ttf lmr_font_sans_bold)
embed_font(FONT_C DejaVuSerif.ttf lmr_font_serif)
embed_font(FONT_C DejaVuSansMono.ttf lmr_font_mono)
file(WRITE "${GEN}/fontdata.c" "${FONT_C}")

set(GP_SRC alloc amos_airy axis boundary breaders color command complexfun contour
  datablock datafile dynarray encoding eval external filters fit gadgets getcolor
  gplocale graph3d graphics help hidden3d history internal interpol jitter libcerf
  loadpath matrix misc mouse multiplot parse plot plot2d plot3d pm3d readline save
  scanner set show specfun standard stats stdfn tables tabulate term time unset util
  util3d version voxelgrid vplot watch xdg bitmap)
list(TRANSFORM GP_SRC PREPEND "${GP}/src/")
list(TRANSFORM GP_SRC APPEND ".c")
add_library(gnuplot_core STATIC ${GP_SRC} cpp/raster/lmraster.cpp cpp/gp_memory.cpp "${GEN}/fontdata.c")
set_source_files_properties(cpp/raster/lmraster.cpp PROPERTIES COMPILE_OPTIONS "-w;-O2")
target_include_directories(gnuplot_core PRIVATE cpp/gpconfig cpp/raster vendor "${GP}/src" "${GP}/term" "${GEN}" cpp)
target_compile_definitions(gnuplot_core PRIVATE HAVE_CONFIG_H LMPLOT_EMBED LMPLOT_VERSION="${PROJECT_VERSION}")
# Memory budget: gnuplot's malloc/free go through counting wrappers (cpp/gp_memory.h, force-included
# into the C sources); only gp_alloc/gp_realloc in alloc.c may be refused (NULL -> "out of memory").
target_compile_options(gnuplot_core PRIVATE "$<$<COMPILE_LANGUAGE:C>:-include>"
  "$<$<COMPILE_LANGUAGE:C>:${CMAKE_CURRENT_SOURCE_DIR}/cpp/gp_memory.h>")
set_source_files_properties("${GP}/src/alloc.c" PROPERTIES COMPILE_DEFINITIONS LMPLOT_MEM_CHECKED)
# plot.c's main() becomes the full gnuplot command line, reachable as `lmplot gnuplot ...`.
set_source_files_properties("${GP}/src/plot.c" PROPERTIES COMPILE_DEFINITIONS "main=gnuplot_cli_main")
if(NOT MSVC)
  target_compile_options(gnuplot_core PRIVATE -w)
endif()
if(NOT WIN32)
  target_link_libraries(gnuplot_core PUBLIC m)
endif()

add_executable(lmplot cpp/main.cpp cpp/memory_budget.cpp)
target_compile_definitions(lmplot PRIVATE LMPLOT_VERSION="${PROJECT_VERSION}")
target_include_directories(lmplot PRIVATE cpp "${GEN}")
target_link_libraries(lmplot PRIVATE gnuplot_core)
target_compile_options(lmplot PRIVATE -Wall -Wextra -Wpedantic)
if((CMAKE_CXX_COMPILER_ID MATCHES "GNU|Clang") AND NOT APPLE)
  target_link_options(lmplot PRIVATE -static-libstdc++ -static-libgcc)
  if(WIN32)
    target_link_options(lmplot PRIVATE -static)
  endif()
  if(CMAKE_BUILD_TYPE STREQUAL "Release")
    target_link_options(lmplot PRIVATE -s)
  endif()
endif()
